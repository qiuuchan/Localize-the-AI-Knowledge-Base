```python
def create_session(
    title: Optional[str] = None,
    session_id: Optional[str] = None,
    history_limit: int = 50,
    db_path: Optional[Path] = None,
) -> str:
    if session_id is None:
        session_id = datetime.now(timezone.utc).strftime("%Y%m%d%H%M%S%f")
    now = _now_iso()
    if not title:
        title = "新会话"
    conn = get_connection(db_path)
    try:
        conn.execute(
            """
            INSERT OR IGNORE INTO sessions
                (session_id, title, created_at, last_active, history_limit)
            VALUES (?, ?, ?, ?, ?)
            """,
            (session_id, title, now, now, history_limit),
        )
        conn.commit()
    finally:
        conn.close()
    return session_id
```

注意:`get_connection` 必须支持 `db_path` 参数。检查当前实现(`backend/core/sqlite.py:21`),已有 `db_path: Optional[Path] = None` 参数,无需改。

```python
from pydantic import BaseModel, Field

class UpdateSessionRequest(BaseModel):
    history_limit: Optional[int] = Field(default=None, ge=1, le=100)
    title: Optional[str] = Field(default=None, max_length=256)


@router.patch("/sessions/{session_id}")
def update_session_endpoint(
    session_id: str, body: UpdateSessionRequest
) -> Dict[str, Any]:
    s = get_session(session_id)
    if not s:
        raise HTTPException(status_code=404, detail="session 不存在")
    fields = []
    params: List[Any] = []
    if body.history_limit is not None:
        fields.append("history_limit = ?")
        params.append(body.history_limit)
    if body.title is not None:
        fields.append("title = ?")
        params.append(body.title)
    if fields:
        params.append(session_id)
        conn = get_connection()
        try:
            conn.execute(
                f"UPDATE sessions SET {', '.join(fields)} WHERE session_id = ?",
                params,
            )
            conn.commit()
        finally:
            conn.close()
    return get_session(session_id)
```
